# Apache Hive Commands

# Open Hive
hive

# Describe the retail sales table
DESCRIBE retail_sales_raw;

# Count total records
SELECT COUNT(*) AS total_records
FROM retail_sales_raw;

# Display distinct regions
SELECT DISTINCT region
FROM retail_sales_clean;

# Calculate product-wise revenue
SELECT product_name,
       SUM(revenue) AS total_revenue
FROM retail_sales_clean
GROUP BY product_name
ORDER BY total_revenue DESC
LIMIT 10;

# Calculate revenue for East region
SELECT region,
       SUM(revenue)
FROM retail_sales_clean
WHERE region = 'East'
GROUP BY region;
